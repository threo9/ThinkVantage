cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(ThinkVantage VERSION 0.7.1 LANGUAGES CXX)
if(DEFINED ENV{THINKVANTAGE_MSVC_INCLUDE_PREFIX})
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "$ENV{THINKVANTAGE_MSVC_INCLUDE_PREFIX}")
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ThinkVantage requires Windows x64 and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(TV_OPTIMIZE_SIZE "Enable size optimization and link-time code generation" ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
get_target_property(qtCoreType Qt6::Core TYPE)
if(NOT qtCoreType STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "Static Qt is required. Run scripts/build-qt-static.ps1 or supply a static Qt prefix.")
endif()
add_library(protocol STATIC src/protocol.cpp)
target_link_libraries(protocol PUBLIC Qt6::Core)
target_include_directories(protocol PUBLIC src)
add_executable(ThinkVantage WIN32 src/main.cpp src/backend.cpp src/window.cpp src/window_ui.cpp src/vendor.cpp src/vendor_data.cpp src/event_hub.cpp src/automation.cpp src/automation_ui.cpp src/operation_journal.cpp src/input.cpp src/tray.cpp src/instance.cpp src/mode_slider.cpp src/ui_update.cpp assets/app.rc)
set_source_files_properties(assets/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.manifest")
target_link_libraries(ThinkVantage PRIVATE protocol Qt6::Widgets Qt6::Network wbemuuid ole32 oleaut32 advapi32 powrprof shell32 uuid version)
target_compile_definitions(ThinkVantage PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(ThinkVantage PRIVATE /utf-8 /W4 /permissive-)
target_compile_options(protocol PRIVATE /utf-8 /W4)
target_link_options(ThinkVantage PRIVATE /MANIFEST:NO /INCREMENTAL:NO /OPT:REF /OPT:ICF)
qt_import_plugins(ThinkVantage NO_DEFAULT INCLUDE Qt6::QWindowsIntegrationPlugin)
if(TV_OPTIMIZE_SIZE)
  string(REGEX REPLACE "(/|-)O2" "/O1" CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE}")
  set_property(TARGET ThinkVantage protocol PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  target_compile_options(ThinkVantage PRIVATE /Gy /Gw)
  target_compile_options(protocol PRIVATE /Gy /Gw)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(protocol_tests tests/protocol_tests.cpp)
  target_link_libraries(protocol_tests PRIVATE protocol Qt6::Test)
  target_compile_options(protocol_tests PRIVATE /utf-8)
  add_test(NAME protocol COMMAND protocol_tests)
  add_executable(ui_tests tests/ui_tests.cpp src/mode_slider.cpp src/ui_update.cpp)
  target_include_directories(ui_tests PRIVATE src)
  target_link_libraries(ui_tests PRIVATE Qt6::Widgets Qt6::Test)
  target_compile_options(ui_tests PRIVATE /utf-8 /W4)
  qt_import_plugins(ui_tests NO_DEFAULT INCLUDE Qt6::QWindowsIntegrationPlugin Qt6::QOffscreenIntegrationPlugin)
  add_test(NAME ui_updates COMMAND ui_tests)
  set_tests_properties(ui_updates PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(vendor_tests tests/vendor_tests.cpp src/vendor_data.cpp)
  target_include_directories(vendor_tests PRIVATE src)
  target_link_libraries(vendor_tests PRIVATE Qt6::Test)
  target_compile_options(vendor_tests PRIVATE /utf-8 /W4)
  add_test(NAME vendor_data COMMAND vendor_tests)
  add_executable(refresh_safety_tests tests/refresh_safety_tests.cpp)
  target_include_directories(refresh_safety_tests PRIVATE src)
  target_link_libraries(refresh_safety_tests PRIVATE Qt6::Test)
  target_compile_options(refresh_safety_tests PRIVATE /utf-8 /W4)
  add_test(NAME refresh_safety COMMAND refresh_safety_tests)
  add_executable(automation_tests tests/automation_tests.cpp src/automation.cpp)
  target_include_directories(automation_tests PRIVATE src)
  target_link_libraries(automation_tests PRIVATE protocol Qt6::Test)
  target_compile_options(automation_tests PRIVATE /utf-8 /W4)
  add_test(NAME automation COMMAND automation_tests)
  add_executable(target_context_tests tests/target_context_tests.cpp)
  target_include_directories(target_context_tests PRIVATE src)
  target_link_libraries(target_context_tests PRIVATE Qt6::Test)
  target_compile_options(target_context_tests PRIVATE /utf-8 /W4)
  add_test(NAME target_context COMMAND target_context_tests)
  add_executable(handoff_tests tests/handoff_tests.cpp src/instance.cpp)
  target_include_directories(handoff_tests PRIVATE src)
  target_link_libraries(handoff_tests PRIVATE Qt6::Test Qt6::Network shell32)
  target_compile_options(handoff_tests PRIVATE /utf-8 /W4)
  add_test(NAME handoff COMMAND handoff_tests)
endif()
